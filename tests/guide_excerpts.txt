        if kind == "turn_start":
            self._turn_start()
        elif kind == "message_start" and self._open and rec["message"].get("role") == "assistant":
            self._stream_t, self._wrote = time.perf_counter(), False

            self._tool_start(rec["toolCallId"], rec["toolName"], rec.get("args") or {})
        elif kind == "tool_execution_end" and rec.get("toolCallId") in self._calls and not rec.get("parentToolCallId"):
            self._tool_end(rec["toolCallId"], rec.get("result") or {}, rec.get("isError", False), rec.get("durationMs"))
        elif kind == "extension_ui_request":
            self._dialog(rec)
        return kind == "agent_settled"

    class Family:
        key: str
        name: str
        model_types: tuple[str, ...]  # config.json `model_type` values of this family

        # Reading the model's output.
        parse: Callable[..., dict]  # (raw output, {tool name: parameters schema}, language of its errors) -> thinking, tool_calls, content
        thought_prefix: str  # a prompt that ends with this has the model start inside its thinking
        channel_switch: dict[str, str]  # tokens that switch what the model is writing: thought, tool_call, content
        stop_tokens: dict[str, dict[str, str]]  # tokens that end a generation -> language -> what stopping there means

        # Writing the conversation back for the chat template.
        reasoning_key: str  # the assistant-message field the chat template reads earlier thinking from
        text_with_calls: bool  # whether text written before a tool call survives re-rendering the history

        # Showing it on the page; the text is in both of the page's languages (i18n.py).
        user_turn: str  # where a user message starts in a prompt; the step card shows the newest one
        special_groups: dict[str, list]  # language -> (title, subtitle, ((token, name, what it is for), ...)) for each group of control tokens
        thinking_help: dict[str, str]  # what the thinking toggle does to the prompt
        tools_note: dict[str, str]  # how the tool schemas appear in the prompt
        model_note: dict[str, str] = field(default_factory=dict)
        sampling: dict | None = None  # recommended temperature / top_p / top_k when a checkpoint ships none
        byte_level: bool = False  # the vocabulary spells bytes as characters, GPT-2 style (BYTE_OF); otherwise it is SentencePiece's
        processor_kwargs: dict = field(default_factory=dict)  # passed to the processor along with images, audio and video
        file_notes: dict[str, FileNote] = field(default_factory=dict)  # a file in the checkpoint's folder -> what it holds in this family
        drafter_file_notes: dict[str, FileNote] = field(default_factory=dict)  # the same for the folder of its MTP drafter (DRAFTERS)

        def show_piece(self, piece: str) -> str:
            """A vocabulary entry as the text it stands for, which is what a token's popover shows: ␣ for a space,
            ↵ for a newline, and a byte that is only part of a character as <0xE5>, the way SentencePiece writes one."""
            if self.byte_level and all(c in BYTE_OF for c in piece):
                # surrogateescape keeps each byte that is not whole UTF-8 as U+DC80–U+DCFF, so it can be written in hex.
                text = bytes(BYTE_OF[c] for c in piece).decode("utf-8", errors="surrogateescape")
                text = "".join(f"<0x{ord(c) - 0xDC00:02X}>" if "\udc80" <= c <= "\udcff" else c for c in text)
            else:
                text = piece.replace("▁", " ")  # SentencePiece marks a leading space with ▁
            return text.replace(" ", "␣").replace("\n", "↵")

        @cached_property
        def special_tokens(self) -> dict[str, dict[str, tuple[str, str]]]:
            """Control token -> language -> (name, what it is for)."""
            out: dict[str, dict[str, tuple[str, str]]] = {}
            for lang, groups in self.special_groups.items():
                for _, _, rows in groups:
                    for piece, name, desc in rows:
                        out.setdefault(piece, {})[lang] = (name, desc)
            return out

        @cached_property
        def special_re(self) -> re.Pattern:
            """Matches the control tokens in a prompt, longest first so <|image|> is not read as <|image>."""
            pieces = sorted(self.special_tokens, key=len, reverse=True)
            return re.compile("|".join(re.escape(p) for p in pieces))

    def _download(model_id: str, only: str | None = None) -> str:
        """The model's folder in MODELS_DIR, after fetching the files it lacks (only `only`, if given), at the revision
        families.REVISIONS pins it to.

        from_pretrained(local_files_only=True) is no test of the cache: once config.json is cached, a tokenizer whose
        own files are missing raises a ValueError (Gemma 4) or loads empty (Qwen3.5) rather than an OSError.
        snapshot_download lists the model's files on the Hub and fetches only those the cache lacks; when the Hub
        can't be reached it returns the cached folder, so a downloaded model still starts offline."""
        if os.path.isdir(model_id):  # a checkpoint on disk rather than a Hugging Face id
            return model_id
        return snapshot_download(model_id, revision=families.REVISIONS.get(model_id), allow_patterns=only, cache_dir=MODELS_DIR)

    def start_run(who: "Backend | PiAgent", text: str, uploads: list[tuple[str, bytes]] = ()) -> bool:
        """Ask the question in a new run, with `uploads` (file name, contents) saved in the conversation's folder first."""
        slot = generation_slot()
        if not slot.lock.acquire(blocking=False):
            st.toast(tr("另一個 agent 正在執行（可能在別的分頁），請稍等。", "Another agent is running (maybe in another tab). Please wait."))
            return False
        try:
            attachments = [media.save(chat_folder(), name, data, who.modalities) for name, data in uploads]
            run = LiveRun(harness_for(who, text, attachments), text, attachments, slot)
        except Exception:
            slot.lock.release()
            raise
        st.session_state.runs.append(run)
        st.session_state.view = len(st.session_state.runs) - 1
        return True

    class LiveRun:
        """One user turn: the harness thread and the events it has emitted so far. `make` makes the harness (simple_harness.SimpleHarness
        or pi_harness.PiHarness) given the function it sends its events to."""

        def __init__(self, make: Callable[[Callable[[dict], None]], Harness], text: str, attachments: list[media.Attachment], slot: Slot):
            self.text = text
            self.attachments = attachments
            self.events: list[dict] = []
            self.absorbed = False
            self.run = make(self.events.append)
            self._slot = slot
            slot.run = self
            self.thread = threading.Thread(target=self._work, daemon=True)
            self.thread.start()

        def _work(self):
            try:
                self.run.run()
            finally:
                if self._slot.run is self:
                    self._slot.run = None  # do not keep the model alive after a switch to another one

    def encode(self, prompt: str, parts: list[dict]) -> Encoded:
        """The prompt as the model reads it. Without media that is the prompt's tokens. With media (`parts`, the
        conversation's media in the order of their placeholders), the processor replaces each placeholder with the
        tokens that image, clip or video takes, and makes the inputs of the encoders whose output goes there."""
        if not parts:
            ids = self.tokenizer(prompt, add_special_tokens=False, return_tensors="pt").input_ids
            return Encoded(prompt, ids, ids.shape[-1])
        files = {kind: [p["path"] for p in parts if p["type"] == kind] for kind in media.MEDIA}
        inputs = {}
        if files["image"]:
            inputs["images"] = [media.load_image(path) for path in files["image"]]
        if files["audio"]:
            inputs["audio"] = [media.load_audio(path, self.processor.feature_extractor.sampling_rate) for path in files["audio"]]
        if files["video"]:
            videos = [media.load_video(path) for path in files["video"]]
            inputs["videos"] = [frames for frames, _ in videos]
            inputs["videos_kwargs"] = {"video_metadata": [metadata for _, metadata in videos]}
            # A processor that takes a set number of frames (Gemma 4: 32) refuses a video with fewer; it gets all there are.
            wanted, fewest = getattr(self.processor.video_processor, "num_frames", None), min(len(frames) for frames, _ in videos)
            if wanted and fewest < wanted:
                inputs["videos_kwargs"]["num_frames"] = fewest
        out = self.processor(text=prompt, return_tensors="pt", **inputs, **self.family.processor_kwargs)
        ids = out.pop("input_ids")
        out.pop("attention_mask", None)  # one sequence, no padding: every position counts, as with text
        counts = {kind: int((ids == token).sum()) for kind, token in self.media_token_ids.items() if token is not None}
        text = self.tokenizer.decode(ids[0], skip_special_tokens=False)
        return Encoded(text, ids, ids.shape[-1], dict(out.to(self.device, dtype=self.dtype)), {k: n for k, n in counts.items() if n})

    def _encoded_media(self, inputs: dict) -> dict:
        """Run the vision encoder over the prompt's images and videos before the prefill, as generate() does, and pass
        its output in their place. Leaving it to the forward pass breaks videos on Gemma 4: transformers 5.20's
        Gemma4ForConditionalGeneration takes num_frames_per_video but does not hand it on to the video encoder."""
        if not inputs or "mm_encoder_outputs" not in inspect.signature(self.model.forward).parameters:
            return inputs
        base, encoded = self.model.base_model, {}
        for kind in ("image", "video"):
            encoder = getattr(base, f"get_{kind}_features", None)
            args = {k: inputs[k] for k in inspect.signature(encoder).parameters if k in inputs} if encoder else {}
            if any(k.startswith("pixel_values") for k in args):
                encoded[kind] = encoder(**args, return_dict=True)
        if not encoded:
            return inputs
        # The pixels go; the rest stays, such as the grid sizes Qwen3.5's positions are computed from.
        return {**{k: v for k, v in inputs.items() if not k.startswith("pixel_values")}, "mm_encoder_outputs": encoded}

harness/backends/simple_backend.py: def _sample(self, logits
    def _sample(self, logits: torch.Tensor, s: Sampling, draft: int | None = None) -> tuple[int, dict]:
        """Pick the next token and report the model's distribution and what the settings did to it.
        A `draft` (the MTP drafter's guess) is kept with the probability the settings give it, which in greedy
        decoding means when it is the top token; otherwise the pick comes from the other candidates. Either way
        each token comes out exactly as often as without a drafter."""
        probs = torch.softmax(logits, -1)  # the model's own distribution (T = 1, nothing cut off)
        # Equally likely tokens go lowest id first, the one argmax and so generate() takes. topk leaves their order open
        # (on the CPU it can list the higher id first), and in bfloat16 two logits are equal more often than one would think.
        top = sorted(zip(*probs.topk(TOP_N)), key=lambda c: (-float(c[0]), int(c[1])))

        if s.temperature <= 0:  # T -> 0 is greedy decoding
            chosen = int(top[0][1])
            kept = 1
            p_sampled = 1.0
        else:
            scaled = logits / s.temperature
            if s.top_k > 0:
                cand_logits, cand_ids = scaled.topk(min(s.top_k, scaled.numel()))
            else:
                cand_logits, cand_ids = scaled.sort(descending=True)
            cand_probs = torch.softmax(cand_logits, -1)
            # Nucleus: keep the smallest prefix whose probability reaches top_p (always at least one token).
            kept = int(((cand_probs.cumsum(-1) - cand_probs) < s.top_p).sum())
            cand_ids, cand_probs = cand_ids[:kept], cand_probs[:kept] / cand_probs[:kept].sum()
            weights = cand_probs
            at = (cand_ids == draft).nonzero().flatten() if draft is not None else []
            if len(at) and float(torch.rand(())) < float(cand_probs[at[0]]):
                pick = int(at[0])
            else:
                if len(at):  # the guess was turned down: draw from the other candidates
                    weights = cand_probs.clone()
                    weights[at[0]] = 0
                pick = int(torch.multinomial(weights, 1))
            chosen = int(cand_ids[pick])
            p_sampled = float(cand_probs[pick])

        p = float(probs[chosen])
        # Candidates are a prefix of the tokens sorted by probability, so rank decides who survived.
        return chosen, {
            "p": p,
            "logprob": float(torch.log(probs[chosen])) if p > 0 else None,
            "rank": int((probs > probs[chosen]).sum()) + 1,
            "p_sampled": p_sampled,
            "kept": kept,
            "top": [
                {"id": int(i), "piece": self.piece(int(i)), "p": float(v), "kept": rank < kept}
                for rank, (v, i) in enumerate(top)
            ],
        }

    def _details(self, token_id: int, logprobs: dict, n: int, s: Sampling) -> dict:
        """What /v1/completions said about the n-th token of a chunk: its log probability, and those of the 5 most likely
        tokens, keyed "token_id:<id>". The token's rank is known only when it is one of those 5."""
        token_logprobs, top_logprobs = logprobs.get("token_logprobs") or [], logprobs.get("top_logprobs") or []
        if n >= len(token_logprobs) or token_logprobs[n] is None:
            return {}
        logprob = token_logprobs[n]
        listed = top_logprobs[n] if n < len(top_logprobs) and top_logprobs[n] else {}

        listed = {int(k.removeprefix("token_id:")): v for k, v in listed.items() if k.startswith("token_id:")}
        top = sorted(listed.items(), key=lambda c: (-c[1], c[0]))[:TOP_N]
        rank = 1 + sum(v > logprob for _, v in top) if any(i == token_id for i, _ in top) else None
        return self._token_details(token_id, logprob, listed, rank, None, s)

    def _token_details(self, token_id: int, logprob: float, listed: dict[int, float], rank: int | None,
                       mask: list[int] | None, s: Sampling) -> dict:
        """The details of a token event, from what the server said about the token: its log probability, those of the
        tokens listed with it, its rank and its sampling mask when it sent them. What the numbers may be shown as
        depends on the server's --logprobs-mode. Only the model's own log probabilities are p, logprob and the top 5.
        Processed ones are the distribution the token was drawn from, so the token's is its sampling probability, and
        the tokens top-k and top-p cut come back as UNREPORTED. Logits are neither."""
        details = {"rank": rank} if rank else {}
        if self.logprobs_mode == "raw_logprobs":
            # Equally likely tokens go lowest id first, as in SimpleBackend.
            ranked = sorted(listed.items(), key=lambda c: (-c[1], c[0]))
            if rank is not None and rank > TOP_N:  # listed only because it was chosen
                ranked = [c for c in ranked if c[0] != token_id]
            top = [{"id": i, "piece": self.piece(i), "p": math.exp(v)} for i, v in ranked[:TOP_N]]
            details.update(p=math.exp(logprob), logprob=logprob, top=top)
        elif self.logprobs_mode == "processed_logprobs":
            details["p_sampled"] = math.exp(logprob)
        if mask is not None:
            details["kept"] = len(set(mask))
        elif self.logprobs_mode.startswith("processed") and any(v <= UNREPORTED for v in listed.values()):
            # Some of the most likely tokens were cut, and those left are more likely still, so all of them are listed.
            details["kept"] = sum(v > UNREPORTED for v in listed.values())
        return self._greedy(details, token_id, s)

harness/backends/simple_backend.py: def _draft(
    def _draft(self, token_id: int, hidden: torch.Tensor, shared: dict, position: int) -> list[int]:
        """The drafter's greedy guesses for the tokens after `token_id`. It has no cache of its own: it attends to the
        model's keys and values, so every guess sits at the position of `token_id`, and what it has guessed so far
        comes in through its input, the embedding of the last guess next to the drafter's last hidden state."""
        embed = self.model.get_input_embeddings()
        positions = torch.tensor([[position]], device=self.device)
        guesses = []
        while len(guesses) < self.draft_tokens and token_id not in self.eos_ids:
            x = torch.cat([embed(torch.tensor([[token_id]], device=self.device)), hidden], -1)
            out = self.drafter(inputs_embeds=x, position_ids=positions, shared_kv_states=shared)
            token_id, hidden = int(out.logits[0, -1].argmax()), out.last_hidden_state
            guesses.append(token_id)
        return guesses

harness/backends/simple_backend.py: def _drafted(
    def _drafted(self, out, s: Sampling, stats: dict) -> Iterator[tuple[int, dict]]:
        """MTP decoding. After each token the drafter guesses the next few, and one forward pass over the token and the
        guesses gives the model's distribution at every guessed position. _sample checks the guesses in order: each
        one it keeps is a token for free, the first one it turns down is replaced by the model's own pick, and the
        rest are dropped. When every guess holds, the last position gives one more token."""
        past = out.past_key_values
        past.activate_past_recording()  # so the sliding-window layers can still drop guesses that were turned down
        hidden, shared = out.hidden_states[-1][:, -1:], out.shared_kv_states
        token_id, details = self._sample(out.logits[0, -1].float().cpu(), s)
        while True:
            yield token_id, details
            guesses = self._draft(token_id, hidden, shared, past.get_seq_length())
            out = self._forward(torch.tensor([[token_id, *guesses]], device=self.device), past, len(guesses) + 1, True)
            logits = out.logits[0].float().cpu()
            stats["forwards"] += 1
            stats["drafted"] += len(guesses)
            for n, guess in enumerate(guesses):
                token_id, details = self._sample(logits[n], s, guess)
                if token_id != guess:
                    details["draft"] = {"piece": self.piece(guess), "accepted": False}
                    break
                stats["accepted"] += 1
                yield token_id, {**details, "draft": {"piece": self.piece(guess), "accepted": True}}
            else:
                n = len(guesses)
                token_id, details = self._sample(logits[n], s)
            # Keep the cache and the drafter's inputs up to the last kept guess (or the token, when none held).
            dropped = len(guesses) - n
            past.crop(-dropped)
            hidden = out.hidden_states[-1][:, n : n + 1]
            shared = {k: (key[:, :, : key.shape[2] - dropped], value[:, :, : value.shape[2] - dropped])
                      for k, (key, value) in out.shared_kv_states.items()}

harness/backends/vllm_backend.py: def drafted_tokens(
    def drafted_tokens(accepted: list[int], drafted: list[int], n_tokens: int) -> dict | None:
        """Which of a step's tokens were guesses the model kept (kept_at) and which replace a guess it turned down
        (turned_down_at), from vLLM's per-step counts (--per-request-spec-decode-metrics detailed). The first token comes
        from the prefill; after it, each verify step gives the guesses it kept and then one token of the model's own,
        which replaces the next guess when it kept fewer than were drafted. vLLM does not count a step without guesses,
        so when the steps do not account for every token, nobody can tell which tokens were guessed: None."""
        kept, turned_down, at = [], [], 1
        for n_kept, n_drafted in zip(accepted, drafted):
            kept += range(at, at + n_kept)
            if n_kept < n_drafted:
                turned_down.append(at + n_kept)
            at += n_kept + 1
        last = accepted[-1] + 1 if accepted else 1
        if not at - last < n_tokens <= at:  # only a stop token or the token limit may cut the last step short
            return None
        return {"kept_at": [i for i in kept if i < n_tokens], "turned_down_at": [i for i in turned_down if i < n_tokens]}

        """Check arguments against the schema; return them with defaults filled in, plus any errors in `lang`."""
        props = self.parameters.get("properties", {})
        errors = [pick(lang, f"缺少必要參數 {k}", f"Missing required argument {k}") for k in self.parameters.get("required", []) if k not in args]

        filled.update(args)
        return filled, errors

harness/tools.py: def _eval(node):
    def _eval(node):
        if isinstance(node, ast.Expression):
            return _eval(node.body)
        if isinstance(node, ast.Constant) and isinstance(node.value, (int, float)) and not isinstance(node.value, bool):
            return node.value
        if isinstance(node, ast.Name) and node.id in CONSTS:
            return CONSTS[node.id]
        if isinstance(node, ast.BinOp) and type(node.op) in BIN_OPS:
            left, right = _eval(node.left), _eval(node.right)
            # Check the size before computing: ((10**999)**999)**999 would otherwise hang.
            if isinstance(node.op, ast.Pow) and right > 0 and _digits(left) * right > MAX_DIGITS:
                raise ToolError(f"結果超過 {MAX_DIGITS} 位數", f"The result has more than {MAX_DIGITS} digits")
            return BIN_OPS[type(node.op)](left, right)
        if isinstance(node, ast.UnaryOp) and type(node.op) in UNARY_OPS:
            return UNARY_OPS[type(node.op)](_eval(node.operand))
        if isinstance(node, ast.Call) and isinstance(node.func, ast.Name) and node.func.id in FUNCS and not node.keywords:
            args = [_eval(a) for a in node.args]
            if node.func.id == "factorial" and args and args[0] > 500:
                raise ToolError("factorial 的參數太大", "The argument of factorial is too large")
            # round(5, -10**9) computes 10 ** 10**9 inside int.__round__, where the check on powers above cannot see it.
            if node.func.id == "round" and len(args) > 1 and abs(args[1]) > MAX_DIGITS:
                raise ToolError("round 的位數參數太大", "The digits argument of round is too large")
            return FUNCS[node.func.id](*args)
        raise ToolError(f"不支援的運算：{ast.unparse(node)}", f"Unsupported operation: {ast.unparse(node)}")
